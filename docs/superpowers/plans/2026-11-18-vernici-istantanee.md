# Instant Paint Switching Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Picking another paint in the viewer swaps the materials of the model already on screen (no reload, no loading screen), with three colour swatches per paint in the menu and a "Compare with stock paint" button.

**Architecture:** For models with 2+ paints the server builds the viewer's GLB with parts grouped by a paint-independent key (`.mtl` path + sub-material index + reference material), so every paint has the same geometry. A new `/api/model/paint` returns each group's material in a paint, written exactly as the GLB would write it; `web/paint.js` rebuilds those materials (through `GameLook.build`) and swaps them in place. Interiors are built once and reused by every paint. Any paint that fails the safety check falls back to today's reload.

**Tech Stack:** Python 3.12 / FastAPI / trimesh / numpy (server), vanilla JS + three.js r160 (viewer), pytest, headless Edge driven over the DevTools protocol (`tools/`).

**Spec:** `docs/superpowers/specs/2026-11-18-vernici-istantanee-design.md` (Italian). Read it first.

```python
def _q(values):
    """Colour factors as trimesh stores them in the GLB (uint8): the viewer's paint tables must match bit for bit."""
    return [float(np.round(np.clip(float(c), 0.0, 1.0) * 255.0) / 255.0) for c in values]


def keyed_form(mat):
    """(material, off) for a keyed GLB (paints.py). A livery stencil decal is hidden when its paint has no atlas
    (materials.py); in a keyed GLB it keeps one shape in every paint: a decal with UVs, `off` (never drawn) here."""
    if mat.stencil_decal and mat.kind == "hidden":
        return replace(mat, kind="decal"), True
    return mat, False


def glb_material(mat, has_uv, textures, tex_size, lite=False, group=None, off=False):
    """The glTF material _glb_mesh writes for `mat`: (fields, base image | None, normal image | None, uv scale,
    shading | None). `fields` is the JSON form the viewer rebuilds when it swaps paints (paints.paint_table):
    {name, color, alpha, metallic, roughness, emissive, alpha_mode, alpha_cutoff, texture, normal_texture, visible}.
    group: paint group of a keyed GLB, written into the name; off: see keyed_form."""
    kind = mat.kind
    sh = materials.shading(mat) if lite and not off else None
    if off:
        base, normal, uv_scale = None, None, 1.0
    elif sh and not mat.stencil_colors:
        base, normal, uv_scale = None, None, sh["uv_scale"]
    else:
        base, normal, uv_scale = material_textures(mat, textures, tex_size) if has_uv else (None, None, 1.0)
    rgb = _srgb(mat.color)
    alpha = 1.0
    metallic, rough = (0.85, 0.45) if mat.metal else (0.3, 0.6)
    if kind == "glass":
        # see-through as much as the glass lets light through, smooth and reflective (materials.glass_look)
        # (the body colour goes in linear, as glTF wants: sRGB-encoded it lit up as a pale grey veil)
        body, alpha = materials.glass_look(mat)
        rgb, metallic, rough = np.asarray(body), 0.0, 0.05
    if off:
        alpha_mode, alpha = "BLEND", 0.0
    elif group is not None and mat.stencil_decal:
        alpha_mode = "BLEND"                       # keyed GLB: one shape in every paint (see keyed_form)
    elif base is not None and base.mode == "RGBA":
        alpha_mode = "BLEND" if kind == "decal" else "MASK"
    else:
        alpha_mode = "BLEND" if kind == "glass" else "OPAQUE"
    color = _q([*rgb, alpha])
    fields = {"name": f"{kind}|{mat.name}|{'' if off else _sid(mat)}" + (f"|g{group}" if group is not None else ""),
              "color": color[:3], "alpha": color[3], "metallic": metallic, "roughness": rough,
              "emissive": [float(c) for c in _srgb(mat.emissive)] if mat.emissive else None,
              "alpha_mode": alpha_mode,
              "alpha_cutoff": (mat.alpha_test or 0.5) if alpha_mode == "MASK" else None,
              "texture": base is not None, "normal_texture": normal is not None, "visible": not off}
    return fields, base, normal, uv_scale, sh


def screen_fields(mat, group=None):
    """Name and glow of a screen material as _screen_mesh writes them. The group goes before the |rtt / |holo
    suffix: the viewer finds monitors and HUD veils by that ending (app.js)."""
    tail = "|holo" if mat.holo else "|rtt" if mat.rtt else ""
    glow = [float(c) for c in (mat.emissive or (0.15, 0.6, 0.8))]
    return {"name": f"screen|{mat.name}" + (f"|g{group}" if group is not None else "") + tail, "emissive": glow}
```

```python
def _glb_mesh(mat, v, f, uv, textures, tex_size, axis=True, uv2=None, lite=False, group=None, off=False):
    """lite: for the viewer's game look, which loads the textures itself (/api/texture): materials it rebuilds get
    their UVs but no embedded images (saves encoding and hashing them on every load).
    group / off: keyed GLB (paints.py, keyed_form)."""
    if mat.kind == "screen":
        return _screen_mesh(mat, v, f, uv, axis, group)
    fields, base, normal, uv_scale, sh = glb_material(mat, uv is not None, textures, tex_size, lite, group, off)
    pbr = trimesh.visual.material.PBRMaterial(
        name=fields["name"], baseColorFactor=[*fields["color"], fields["alpha"]], baseColorTexture=base,
        normalTexture=normal, metallicFactor=fields["metallic"], roughnessFactor=fields["roughness"],
        alphaMode=fields["alpha_mode"], alphaCutoff=fields["alpha_cutoff"],
        emissiveFactor=fields["emissive"], doubleSided=True)
    mesh = trimesh.Trimesh(v @ Z_UP_TO_Y_UP.T if axis else v, f, process=False)
    if uv is not None and (base is not None or normal is not None or sh or off):
        # trimesh expects OpenGL UVs (V up) and flips them back when writing glTF
        t = uv.astype(np.float64) * uv_scale
        mesh.visual = trimesh.visual.TextureVisuals(uv=np.column_stack([t[:, 0], 1.0 - t[:, 1]]), material=pbr)
    else:
        mesh.visual = trimesh.visual.TextureVisuals(material=pbr)
    if uv2 is not None and (mat.stencil_map or group is not None):
        # custom glTF attribute (three.js reads it as "_uv2"): the camo/livery stencil is laid out on it. A keyed GLB
        # always carries it: a camo paint can put a stencil on a hull the stock paint leaves plain (Perseus)
        mesh.vertex_attributes["_UV2"] = uv2.astype(np.float32)
    return mesh
```

```python
"""Instant paint switching (web/paint.js). The viewer's GLB of a model with several paints groups its parts by where
their material comes from, which does not depend on the paint, so every paint has the same geometry; a paint is then
just a table of materials, one per group, written as the GLB would write them (export.glb_material)."""
from . import export, materials


def group_keys(ref_model):
    """Group number of every part of `ref_model`, the model in its reference paint (variant_list()[0]). A paint only
    changes a part's material through its .mtl path and sub-material index (model.with_variant), so parts sharing
    both, and their reference material, share their material in every paint. Parts without a .mtl path are never
    repainted (model.with_variant leaves them alone): grouped by material."""
    index, out, memo = {}, [], {}
    for p in ref_model.parts:
        k = memo.get(id(p.material))              # repr() once per material object: interiors have ~60k parts
        if k is None:
            k = memo[id(p.material)] = repr(p.material)
        key = ("mat", k) if not p.mtl_path or p.mat_id < 0 else ("slot", p.mtl_path, p.mat_id, k)
        out.append(index.setdefault(key, len(index)))
    return out


def uv_scale(mat):
    """The repeat the GLB multiplies into a material's UVs (materials.shading, export.material_textures)."""
    if mat.kind not in ("solid", "decal") or not mat.diffuse or abs(mat.tiling - 1.0) <= 1e-3:
        return 1.0
    return 1.0 if (mat.normal and not mat.layer_normal) else mat.tiling


def _look(mat):
    """What has to stay the same for a material to be swapped in place: kind and name (mesh names, switches), UV
    repeat (baked into the GLB) and how the viewer draws it (batches and shadows depend on transparency)."""
    if mat.stencil_decal:
        return ("stencil", mat.name)               # always a BLEND decal in a keyed GLB (export.glb_material)
    if mat.kind == "screen":
        return ("screen", mat.name, mat.holo, mat.rtt)   # the name's |holo / |rtt ending (app.js)
    glows = bool(mat.emissive and any(mat.emissive))
    see = mat.kind == "glass" or (mat.kind == "decal" and (mat.pom or (mat.opacity < 1 and not glows)))
    return (mat.kind, mat.name, uv_scale(mat), see, mat.pom, mat.alpha_test > 0)


def check(ref, model, groups):
    """True when `model` (another paint of the same model, same full/interior) can take the place of `ref` by
    swapping materials only."""
    if len(ref.parts) != len(model.parts):
        return False
    first = {}
    for g, a, b in zip(groups, ref.parts, model.parts):
        if a.node != b.node or len(a.faces) != len(b.faces):
            return False
        mb = b.material
        if g in first:
            if first[g] is not mb and repr(first[g]) != repr(mb):
                return False                       # parts of one group must share their material in this paint too
        else:
            first[g] = mb
            if _look(a.material) != _look(mb):
                return False
    return True


def swatches(palette):
    """The paint's three colours (entries A, B, C, sRGB 0-255) for the menu, or None (no palette, or all black)."""
    cols = [(palette or {}).get(k) for k in "ABC"]
    if not all(cols) or not any(any(c) for c in cols):
        return None
    return [list(c) for c in cols]


def paint_table(model, groups, textures, tex_size=1024, fixed_from=None):
    """{"groups": one entry per group (the material in `model`'s paint as the GLB writes it, or None for groups that
    never change: proxies, and with fixed_from the groups that start at or after that part, the interior's),
    "shading": the game-look description of every shading id used}."""
    out = [None] * ((max(groups) + 1) if groups else 0)
    has_uv = {g for g, p in zip(groups, model.parts) if p.uvs is not None}   # a merged mesh has UVs if any part has
    shading, seen = {}, set()
    for i, (g, p) in enumerate(zip(groups, model.parts)):
        if g in seen:
            continue
        seen.add(g)
        if fixed_from is not None and i >= fixed_from:
            continue
        mat = p.material
        if mat.kind == "screen":
            out[g] = export.screen_fields(mat, g)
            continue
        mat, off = export.keyed_form(mat)
        if mat.kind == "hidden":
            continue
        fields, _, _, _, sh = export.glb_material(mat, g in has_uv, textures, tex_size, lite=True, group=g, off=off)
        out[g] = fields
        if sh and not off:
            shading[materials.shading_id(sh)] = sh
    return {"groups": out, "shading": shading}
```

```js
const KEEP = 3;                                   // paints kept built: the one on screen, stock (Compare), one more
const NO_HOOK = THREE.Object3D.prototype.onBeforeRender;
const kindOf = (name) => name.split("|")[0] || "solid";
const meshKind = (o) => kindOf(o.name || o.parent?.name || "");

// the material GLTFLoader makes from a glTF material of the viewer's lite GLB: what the game look expects as `old`.
// The lite GLB has no normals or tangents, so GLTFLoader makes its materials flat shaded and flips the normal map's
// green channel (GLTFLoader.js assignFinalMaterial); its textures repeat.
function standIn(e, map, normal) {
  const m = new THREE.MeshStandardMaterial({ name: e.name, metalness: e.metallic, roughness: e.roughness, flatShading: true });
  m.color.setRGB(e.color[0], e.color[1], e.color[2]);          // glTF factors are linear, like Color.setRGB
  m.opacity = e.alpha;
  if (e.emissive) m.emissive.setRGB(e.emissive[0], e.emissive[1], e.emissive[2]);
  if (e.alpha_mode === "BLEND") { m.transparent = true; m.depthWrite = false; }
  else if (e.alpha_mode === "MASK") m.alphaTest = e.alpha_cutoff ?? 0.5;
  if (map) m.map = map;
  if (normal) { m.normalMap = normal; m.normalScale.set(1, -1); }
  return m;
}

// three vertices of `g` with every attribute it has (but the batch's own batchId, which BatchedMesh refuses in a
// geometry it is given): compiles a shader's BatchedMesh variant cheaply
function tinyLike(g) {
  const t = new THREE.BufferGeometry();
  for (const [n, a] of Object.entries(g.attributes)) {
    if (n === "batchId") continue;
    t.setAttribute(n, new THREE.BufferAttribute(a.array.slice(0, 3 * a.itemSize), a.itemSize, a.normalized));
  }
  t.setIndex([0, 1, 2]);
  return t;
}

// Builds a paint's materials the way the loader and the game look would, compiles them off screen, then swaps them in
// on one frame. Glass keeps its material: only the colour of its tint layer changes (app.js glassLook). Screens are
// left to app.js (afterPaint), which rebuilds the screen classes that draw on them.
export class PaintSwap {
  // gameLook: the model's GameLook (textures shared with the paint loaded first); batches: its Batches; root: the
  // model; compile(root, section): app.js compileShaders; section: the model has an interior; look: GameLook.build
  // options; query: "path=..&full=..&interior=.." of the model; glassTint([r,g,b]) -> THREE.Color; tag: paint on screen
  constructor({ gameLook, batches, root, compile, section, look, query, glassTint, tag }) {
    Object.assign(this, { gameLook, batches, root, compile, section, look, query, glassTint });
    this.meshes = new Map();                      // group -> meshes in the scene (batched pieces are in `batches`)
    root.traverse((o) => {
      const g = o.userData.group;
      if (!o.isMesh || o.isBatchedMesh || g == null) return;
      if (!this.meshes.has(g)) this.meshes.set(g, []);
      this.meshes.get(g).push(o);
    });
    this.paints = new Map();                      // tag -> { mats: Map(group -> material), tints, textures }, oldest first
    this.current = tag;
    this.seq = 0;
    this.disposed = false;
    this.paints.set(tag, this.snapshot());        // the paint on screen: going back to it costs nothing
  }

  // the materials on screen now, by group (the game material while the resin look is on): only the groups a paint
  // rebuilds (solids and decals); glass keeps its material and only its tint is recorded, screens belong to the
  // screen classes (app.js), proxies never change
  snapshot() {
    const mats = new Map(), tints = new Map();
    const take = (g, o) => { if (!mats.has(g)) mats.set(g, o.userData.gameMaterial || o.material); };
    for (const [g, list] of this.meshes) {
      const kind = meshKind(list[0]);
      if (kind === "solid" || kind === "decal") take(g, list[0]);
      const layer = list[0].children.find((c) => c.userData.glass);
      if (layer) tints.set(g, (layer.userData.gameMaterial || layer.material).color.clone());
    }
    for (const b of this.batches.list) if (b.userData.group != null) take(b.userData.group, b);   // solids only
    return { mats, tints, textures: [], paint: null };
  }

  // the /api/model/paint answer of a kept paint (null for the paint loaded first): no request needed to show it again
  answer(tag) {
    return this.paints.get(tag)?.paint || null;
  }

  // Show paint `tag` (`paint`: the /api/model/paint answer). False when a newer request took over meanwhile.
  async apply(tag, paint) {
    const seq = ++this.seq;
    let built = this.paints.get(tag);
    if (!built) {
      built = await this.build(tag, paint, () => seq !== this.seq || this.disposed);
      if (seq !== this.seq || this.disposed) { this.free(built); return false; }
    }
    built.paint = paint;
    this.assign(built);
    this.current = tag;
    this.paints.delete(tag);
    this.paints.set(tag, built);                  // most recent last
    this.evict();
    return true;
  }

  async build(tag, paint, stale) {
    const mats = new Map(), tints = new Map(), textures = [], slots = [], made = new Map(), jobs = [];
    paint.groups.forEach((e, g) => {
      if (!e) return;
      const kind = kindOf(e.name);
      if (kind === "screen") return;              // app.js afterPaint: the screen classes own those materials
      const sample = this.meshes.get(g)?.[0] || this.batches.byGroup(g)[0];
      if (!sample) return;                        // not drawn in this GLB (proxies, interior-only groups)
      if (kind === "glass") { tints.set(g, this.glassTint(e.color)); return; }
      // groups that look the same share one material (as the game look shares them on load); a group with its own
      // images keeps its own
      const key = JSON.stringify({ ...e, name: e.name.replace(/\|g\d+/, "") }) + (e.texture || e.normal_texture ? `#${g}` : "");
      if (!made.has(key)) made.set(key, this.material(tag, g, e, kind, paint, sample.geometry, slots, textures));
      jobs.push(made.get(key).then((m) => mats.set(g, m)));
    });
    await Promise.all(jobs);
    await this.precompile(mats);
    // full sizes follow, as on load (not for a build a newer request already replaced: it is thrown away)
    if (!stale()) this.gameLook.full = this.gameLook.upgrade([...new Set(mats.values())], slots);
    return { mats, tints, textures, paint };
  }

  async material(tag, g, e, kind, paint, geometry, slots, textures) {
    const [map, normal] = await Promise.all([
      e.texture ? this.image(tag, g, "base", textures) : null,
      e.normal_texture ? this.image(tag, g, "normal", textures) : null]);
    const old = standIn(e, map, normal);
    const sid = e.name.split("|")[2];
    const sh = sid && paint.shading[sid];
    let m = old;
    if (sh && (kind === "solid" || kind === "decal")) {
      const made = await this.gameLook.build(sh, old, kind, this.look, geometry, slots);
      if (made) { m = made; if (made !== old) old.dispose(); }
    }
    m.side = THREE.DoubleSide;                    // as loadModel sets every material
    if (!m.isMeshPhysicalMaterial) m.envMapIntensity = 0.6;
    m.visible = e.visible !== false;              // livery stickers this paint does not have
    return m;
  }

  async image(tag, g, map, textures) {
    const url = `/api/model/paint/texture?${this.query}&variant=${encodeURIComponent(tag)}&group=${g}&map=${map}`;
    try {
      const t = await new THREE.TextureLoader().loadAsync(url);
      t.flipY = false;                            // glTF images, like the GLB's
      t.wrapS = t.wrapT = THREE.RepeatWrapping;   // GLTFLoader's default without a sampler
      if (map === "base") t.colorSpace = THREE.SRGBColorSpace;
      try { this.gameLook.renderer.initTexture(t); } catch { /* uploaded at the first frame instead */ }
      textures.push(t);
      return t;
    } catch {
      return null;                                // no image for this group: drawn without it, as the GLB would
    }
  }

  // compile the new shaders before they are used (one mesh per material, and a tiny BatchedMesh for the materials
  // of batches, whose shaders differ), so the swap itself does not freeze a frame
  async precompile(mats) {
    const stage = new THREE.Group(), batched = new Set();
    try {
      for (const [g, m] of mats) {
        const mesh = this.meshes.get(g)?.[0];
        if (mesh) stage.add(new THREE.Mesh(mesh.geometry, m));
        const b = this.batches.byGroup(g)[0];
        if (b && !batched.has(m)) {
          batched.add(m);
          const tiny = new THREE.BatchedMesh(1, 3, 3, m);
          tiny.addGeometry(tinyLike(b.geometry));
          stage.add(tiny);
        }
      }
      await this.compile(stage, this.section);
    } catch { /* compiled at the first frame instead */ }
    stage.traverse((o) => { if (o.isBatchedMesh) o.dispose(); });
  }

  // castShadow, receiveShadow and the batches (opaque only) were set up for the paint loaded first: the server's
  // swap check (paints.check, condition 5) only lets through paints that draw every group with the same transparency
  assign(built) {
    const put = (o, m) => { if (o.userData.gameMaterial) o.userData.gameMaterial = m; else o.material = m; };
    for (const [g, m] of built.mats) {
      const hook = m.userData.beforeRender;       // POM decals: light direction per frame
      for (const o of this.meshes.get(g) || []) {
        put(o, m);
        o.onBeforeRender = hook ? (r, sc, cam) => hook(cam) : NO_HOOK;
      }
      for (const b of this.batches.byGroup(g)) { put(b, m); b.hook = hook || null; }
    }
    for (const [g, t] of built.tints) {
      for (const o of this.meshes.get(g) || []) {
        const layer = o.children.find((c) => c.userData.glass);
        if (layer) (layer.userData.gameMaterial || layer.material).color.copy(t);
        (o.userData.gameMaterial || o.material).userData.swatch = t;   // material list colour
      }
    }
  }

  evict() {
    while (this.paints.size > KEEP) {
      const [tag, built] = this.paints.entries().next().value;
      if (tag === this.current) break;
      this.paints.delete(tag);
      this.free(built);
    }
  }

  // the materials on screen (game materials while the resin look is on)
  live() {
    const out = new Set();
    for (const list of this.meshes.values()) for (const o of list) out.add(o.userData.gameMaterial || o.material);
    for (const b of this.batches.list) out.add(b.userData.gameMaterial || b.material);
    return out;
  }

  // free a paint that is not on screen; a material still shown (one kept by groups this paint did not rebuild) stays
  free(built) {
    const live = this.live();
    for (const m of new Set(built.mats.values())) if (!live.has(m)) m.dispose();
    for (const t of built.textures) t.dispose();
  }

  // the model is going away: cancel a swap in progress and free the kept paints that are not on screen (loadModel
  // frees the ones on screen with the model)
  dispose() {
    this.disposed = true;
    this.seq++;
    for (const [tag, built] of this.paints) if (tag !== this.current) this.free(built);
    this.paints.clear();
  }
}
```
